// ========================================== // 1. [읽기] 주문 DB, 거래처 DB, 품목 기준단가 동시 반환 // ========================================== function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. 주문 시트 var orderSheet = ss.getSheetByName("주문내역") || ss.getActiveSheet(); var orderRows = orderSheet.getDataRange().getValues(); var orders = []; if (orderRows.length > 1) { for (var i = 1; i < orderRows.length; i++) { var r = orderRows[i]; orders.push({ id: r[0], date: r[1], status: r[2], paymentStatus: r[3] === "입금완료" ? "paid" : "unpaid", clientName: r[4], bizNum: r[5], receiverName: r[6], receiverPhone: r[7], receiverAddress: r[8], memo: r[9], itemsSummary: r[10], totalQty: r[11], supplyPrice: r[12], vat: r[13], totalPrice: r[14], courier: r[15] || "CJ대한통운", trackingNo: r[16] || "", shippingFee: r[17] || 0 }); } orders.reverse(); } // 2. 거래처 시트 var clientSheet = ss.getSheetByName("거래처관리"); var clients = []; if (clientSheet && clientSheet.getLastRow() > 1) { var cRows = clientSheet.getDataRange().getValues(); for (var j = 1; j < cRows.length; j++) { var cr = cRows[j]; var lastPrices = {}; try { lastPrices = cr[6] ? JSON.parse(cr[6]) : {}; } catch(err) {} var addresses = []; try { addresses = cr[7] ? JSON.parse(cr[7]) : []; } catch(err) {} clients.push({ id: cr[0], name: cr[1], bizNum: cr[2], receiverName: cr[3], receiverPhone: cr[4], receiverAddress: cr[5], lastPrices: lastPrices, addresses: addresses }); } } // 3. 품목 기준단가 시트 (없으면 자동 생성) var itemSheet = ss.getSheetByName("품목기준"); var customItems = []; if (!itemSheet) { itemSheet = ss.insertSheet("품목기준"); itemSheet.appendRow(["품목ID", "품목명", "기본단가", "카톤입수"]); var defaultItems = [ ["P01", "참기름 350mL", 7300, 24], ["P02", "들기름 350mL", 8300, 24], ["P03", "참참 선물세트 350mL 2병", 16000, 10], ["P04", "참들 선물세트 350mL 2병", 17000, 10], ["P05", "국산 햇참기름 180mL", 16500, 24], ["P06", "국산 햇들기름 180mL", 14500, 24], ["P07", "국산 참들 선물세트 180mL 2병", 32000, 8], ["P08", "진한 국산 생강청 600g", 8500, 20], ["P09", "안심육수 한알 210g", 6600, 20] ]; defaultItems.forEach(function(row) { itemSheet.appendRow(row); }); } var itemRows = itemSheet.getDataRange().getValues(); for (var k = 1; k < itemRows.length; k++) { customItems.push({ id: itemRows[k][0], name: itemRows[k][1], defaultPrice: Number(itemRows[k][2]), carton: Number(itemRows[k][3]) || 20 }); } return ContentService.createTextOutput(JSON.stringify({ orders: orders, clients: clients, items: customItems })).setMimeType(ContentService.MimeType.JSON); } // ========================================== // 2. [쓰기] 주문 및 거래처 정보 저장 // ========================================== function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.type === "client_sync") { // 거래처 정보 업데이트 var cSheet = ss.getSheetByName("거래처관리") || ss.insertSheet("거래처관리"); if (cSheet.getLastRow() === 0) { cSheet.appendRow(["거래처ID", "상호명", "사업자번호", "대표자/담당자", "연락처", "기본주소", "단가기억(JSON)", "배송지이력(JSON)"]); } var cRows = cSheet.getDataRange().getValues(); var cIdx = -1; for (var i = 1; i < cRows.length; i++) { if (cRows[i][0] == data.client.id || cRows[i][1] == data.client.name) { cIdx = i + 1; break; } } var cData = [ data.client.id, data.client.name, data.client.bizNum || "", data.client.receiverName || "", data.client.receiverPhone || "", data.client.receiverAddress || "", JSON.stringify(data.client.lastPrices || {}), JSON.stringify(data.client.addresses || []) ]; if (cIdx > 0) cSheet.getRange(cIdx, 1, 1, cData.length).setValues([cData]); else cSheet.appendRow(cData); } else { // 주문 저장/수정 var sheet = ss.getSheetByName("주문내역") || ss.insertSheet("주문내역"); if (sheet.getLastRow() === 0) { sheet.appendRow([ "주문번호", "주문일시", "상태", "입금여부", "발주처", "사업자번호", "수령인", "연락처", "배송주소", "배송메모", "품목요약", "총수량", "공급가액", "부가세", "합계금액", "택배사", "송장번호", "택배비" ]); } var rows = sheet.getDataRange().getValues(); var rowIndex = -1; for (var k = 1; k < rows.length; k++) { if (rows[k][0] == data.id) { rowIndex = k + 1; break; } } var rowData = [ data.id, data.date ? data.date.slice(0, 16).replace('T', ' ') : new Date().toLocaleString(), data.status || "접수완료", data.paymentStatus === "paid" ? "입금완료" : "미입금", data.clientName || "", data.bizNum || "", data.receiverName || "", data.receiverPhone || "", data.receiverAddress || "", data.memo || "", data.itemsSummary || "", data.totalQty || 0, data.supplyPrice || 0, data.vat || 0, data.totalPrice || 0, data.courier || "CJ대한통운", data.trackingNo || "", data.shippingFee || 0 ]; if (rowIndex > 0) sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]); else sheet.appendRow(rowData); } return ContentService.createTextOutput(JSON.stringify({ result: "success" })) .setMimeType(ContentService.MimeType.JSON); } // ========================================== // 3. [자동 백업] 매일 자정 구글 드라이브 복사 // ========================================== function createDailyBackup() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var file = DriveApp.getFileById(ss.getId()); var today = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyyMMdd"); var folders = DriveApp.getFoldersByName("까치방앗간_DB백업"); var backupFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder("까치방앗간_DB백업"); file.makeCopy("[자동백업] 까치방앗간_주문DB_" + today, backupFolder); }
